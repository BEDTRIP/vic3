## Задачи

```base
filters:
  and:
    - type == "issue"
    - file.hasLink(this.file)
views:
  - type: table
    name: Открытые
    filters:
      and:
        - status != "done"
        - status != "wontdo"
        - status != "verified"
        - status != "invalid"
        - status != "duplicate"
        - status != "wontfix"
    groupBy:
      property: status
      direction: ASC
    order:
      - file.name
      - question
      - answer
      - description
    sort:
      - property: date
        direction: ASC
    rowHeight: tall
    columnSize:
      file.name: 180
      note.question: 190
      note.answer: 130
      note.description: 200
  - type: table
    name: Архив
    filters:
      or:
        - status == "done"
        - status == "wontdo"
        - status == "verified"
        - status == "invalid"
        - status == "duplicate"
        - status == "wontfix"
    order:
      - file.name
      - status
      - verified_by
      - description
    sort:
      - property: date
        direction: DESC
    rowHeight: tall
    columnSize:
      file.name: 200
      note.status: 90
      note.verified_by: 160
      note.description: 250
```
